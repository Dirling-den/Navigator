cmake_minimum_required(VERSION 3.24)
project(GraphNavigator LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
include(CTest)
option(GRAPHNAVIGATOR_BUILD_APP "Build graphical application" ON)
set(tinyxml2_BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(tinyxml2 URL https://github.com/leethomason/tinyxml2/archive/refs/tags/11.0.0.tar.gz)
FetchContent_MakeAvailable(tinyxml2)
add_library(graph_core Navigator.cpp MapParser.cpp)
target_include_directories(graph_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(graph_core PUBLIC tinyxml2::tinyxml2)
if(MSVC)
    add_compile_options(/utf-8 /W4)
    target_compile_options(graph_core PRIVATE /utf-8 /W4)
endif()
if(GRAPHNAVIGATOR_BUILD_APP)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz)
    FetchContent_MakeAvailable(raylib)
    add_executable(GraphNavigator main.cpp Render.cpp)
    target_link_libraries(GraphNavigator PRIVATE graph_core raylib)
    if(BUILD_TESTING)
        add_executable(app_logic_tests tests/app_logic_tests.cpp)
        target_link_libraries(app_logic_tests PRIVATE graph_core raylib)
        add_test(NAME app_logic_regressions COMMAND app_logic_tests)
    endif()
endif()
# Новый клиент: пакет .gnpk + CSR-роутинг (без графики) и 3D-приложение.
add_library(gn_core client/data/Package.cpp client/routing/Router.cpp)
target_include_directories(gn_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/client)
find_package(Threads REQUIRED)
target_link_libraries(gn_core PUBLIC Threads::Threads)
if(GRAPHNAVIGATOR_BUILD_APP)
    add_executable(GraphNavigatorCity WIN32 client/App.cpp client/render/CityRenderer.cpp client/ui/Widgets.cpp)
    target_link_libraries(GraphNavigatorCity PRIVATE gn_core raylib)
    if(MSVC)
        # WIN32-приложение без консоли, но с обычной точкой входа main().
        target_link_options(GraphNavigatorCity PRIVATE /ENTRY:mainCRTStartup)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(package_tests tests/package_tests.cpp)
    target_link_libraries(package_tests PRIVATE gn_core)
    add_test(NAME package_regressions COMMAND package_tests ${CMAKE_CURRENT_SOURCE_DIR}/maps/demo_district.gnpk)
    add_executable(graph_tests tests/graph_tests.cpp)
    target_link_libraries(graph_tests PRIVATE graph_core)
    add_test(NAME graph_regressions COMMAND graph_tests)
endif()
